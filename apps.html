STEP 1 • YOUR DATABASE & BACKEND
Write a Rust HFS (.wasm) — haltproof-subroutines ↗
Instead of spinning up an EC2 server and a Postgres database, write `#![no_std]` Rust handlers using `hps_sdk::storage`: 
// Replaces Express API + SQL Table:
hps_entry! {
  "postNote(string)" => handle_post,
  "getNotes()"       => handle_list,
}
fn handle_post(r: &CalldataReader) -> Result<(), i32> {
  let msg = r.dynamic_bytes(0).ok_or(-10)?;
  storage::set(b"latest_note", msg); // Atomic L1 DB!
  ret::write_bool(true);
  Ok(())
}
